  <script>
    const NOMES = { normal: "Normal", atencao: "Atenção", cuidado: "Cuidado extremo", perigo: "Perigo" };
    function nome(n) { return NOMES[n] || n || "sem nível"; }
    function celula(texto, nivel) {
      const td = document.createElement("td");
      td.textContent = texto;
      if (nivel) td.className = nivel;
      return td;
    }
    async function atualizar() {
      const corpo = document.getElementById("salas");
      const aviso = document.getElementById("aviso");
      corpo.replaceChildren();
      try {
        const resposta = await fetch("/ambientes");
        if (!resposta.ok) throw new Error("salas");
        const salas = await resposta.json();
        if (!salas.length) {
          const tr = document.createElement("tr");
          const td = document.createElement("td");
          td.colSpan = 4;
          td.textContent = "Nenhuma sala";
          tr.append(td);
          corpo.append(tr);
          return;
        }
        for (const sala of salas) {
          let prev = null;
          try {
            const r = await fetch("/ambientes/" + encodeURIComponent(sala.id) + "/previsao");
            if (r.ok) prev = await r.json();
          } catch (e) { prev = null; }
          const agora = sala.semLeitura ? "sem leitura" : nome(sala.nivelAtual);
          let previsto = "sem previsão";
          if (prev) {
            previsto = nome(prev.nivelPrevisto);
            if (!prev.ruaDisponivel) previsto += " (sem dado da rua: mantém a última leitura válida da sala)";
            else if (prev.semLeiturasRecentes) previsto += " (só a rua, sala sem leituras recentes)";
            if (prev.demonstracao) previsto += " — demonstração";
          }
          const tr = document.createElement("tr");
          tr.append(
            celula(sala.nome),
            celula(agora, sala.semLeitura ? "" : sala.nivelAtual),
            celula(sala.semLeitura ? "—" : Number(sala.indiceAtual).toFixed(1) + " °C"),
            celula(previsto, prev ? prev.nivelPrevisto : "")
          );
          corpo.append(tr);
        }
        aviso.textContent = "Atualizado.";
      } catch (e) {
        const tr = document.createElement("tr");
        const td = document.createElement("td");
        td.colSpan = 4;
        td.textContent = "Não foi possível carregar as salas.";
        tr.append(td);
        corpo.append(tr);
        aviso.textContent = "Falha ao carregar.";
      }
    }
    atualizar();
    setInterval(atualizar, 30000);
  </script>
