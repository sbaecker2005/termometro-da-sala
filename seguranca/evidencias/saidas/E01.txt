24:         var t = tempC * 9 / 5 + 32;

11:     // uma leitura assinada tem cerca de 250 bytes; acima disso a mensagem é descartada antes do JSON

13:     private readonly IServiceScopeFactory _escopos;
14:     private readonly IConfiguration _config;

16: 
17:     public AssinanteMqtt(IServiceScopeFactory escopos, IConfiguration config, ILogger<AssinanteMqtt> log)
18:     {
19:         _escopos = escopos;
20:         _config = config;
21:         _log = log;
22:     }
23: 
24:     protected override async Task ExecuteAsync(CancellationToken stoppingToken)
25:     {
26:         var broker = _config["Mqtt:Broker"] ?? "broker.hivemq.com";
27:         var porta = _config.GetValue("Mqtt:Porta", 1883);
28:         var topico = _config["Mqtt:Topico"] ?? "fiap-gs/termometro-da-sala/leituras";
29:         var fabrica = new MqttFactory();
30:         using var cliente = fabrica.CreateMqttClient();
31:         cliente.ApplicationMessageReceivedAsync += e => Tratar(e.ApplicationMessage.PayloadSegment);
32:         var opcoes = new MqttClientOptionsBuilder()
33:             .WithTcpServer(broker, porta)
34:             .WithClientId("tds-" + Guid.NewGuid().ToString("N")[..12])
35:             .Build();
36: 
37:         while (!stoppingToken.IsCancellationRequested)
38:         {
39:             try
40:             {
41:                 if (!cliente.IsConnected)
42:                 {
43:                     await cliente.ConnectAsync(opcoes, stoppingToken);
44:                     await cliente.SubscribeAsync(fabrica.CreateSubscribeOptionsBuilder().WithTopicFilter(topico).Build(), stoppingToken);
45:                     _log.LogInformation("mqtt conectado em {Broker} {Topico}", broker, topico);
46:                 }
47:             }
48:             catch (Exception ex) when (ex is not OperationCanceledException)
49:             {
50:                 _log.LogWarning("mqtt indisponivel, nova tentativa: {Mensagem}", ex.Message);
51:             }
52: 
53:             try { await Task.Delay(TimeSpan.FromSeconds(5), stoppingToken); }
54:             catch (OperationCanceledException) { break; }
55:         }
56:     }
57: 
58:     public async Task Tratar(ArraySegment<byte> carga)
59:     {
60:         if (carga.Count > MaximoBytes)
61:         {
62:             _log.LogWarning("leitura mqtt ignorada: {Mensagem}", "acima-do-limite");
63:             return;
64:         }
65:         try
66:         {
67:             var entrada = JsonSerializer.Deserialize<LeituraEntrada>(carga.AsSpan(), new JsonSerializerOptions { PropertyNameCaseInsensitive = true });
68:             if (entrada is null) return;
69:             using var escopo = _escopos.CreateScope();
70:             await escopo.ServiceProvider.GetRequiredService<ServicoLeituras>().Receber(entrada);
71:         }
72:         catch (Exception ex)
73:         {
74:             _log.LogWarning("leitura mqtt ignorada: {Mensagem}", ex.Message);
75:         }
76:     }
77: }

81:         if (entrada.Read(marcador, 0, 2) != 2 || marcador[0] != 0xFF || marcador[1] != 0xD8) return [];
82:         saida.Write(marcador, 0, 2);
83:         while (true)
84:         {
85:             if (entrada.Read(marcador, 0, 2) != 2 || marcador[0] != 0xFF) break;
86:             var m = marcador[1];
87:             if (m == 0xD9) return []; // fim antes da varredura: não há imagem
88:             if (m == 0xDA)
89:             {
90:                 if (!temQuadro) return [];
91:                 saida.Write(marcador, 0, 2);
92:                 entrada.CopyTo(saida);
93:                 return saida.ToArray();
94:             }
95:             if (m == 0x01 || m is >= 0xD0 and <= 0xD8)
96:             {
97:                 saida.Write(marcador, 0, 2);
98:                 continue;
99:             }
100:             if (entrada.Read(tam, 0, 2) != 2) break;
101:             var n = (tam[0] << 8) | tam[1];
102:             if (n < 2 || entrada.Length - entrada.Position < n - 2) break;
103:             var dados = new byte[n - 2];
104:             if (entrada.Read(dados, 0, dados.Length) != dados.Length) break;
105:             if (m == 0xE1) continue;
106:             // SOF0..SOF15 (C4 é DHT, C8 é reservado, CC é DAC): precisão, altura (2), largura (2)
107:             if (m is >= 0xC0 and <= 0xCF and not (0xC4 or 0xC8 or 0xCC) && dados.Length >= 5)
108:                 temQuadro = ((dados[1] << 8) | dados[2]) > 0 && ((dados[3] << 8) | dados[4]) > 0;
109:             saida.Write(marcador, 0, 2);
110:             saida.Write(tam, 0, 2);
111:             saida.Write(dados, 0, dados.Length);
112:         }
113:         return [];
114:     }
115: 
116:     private BadRequestObjectResult Recusar(string motivo)
117:     {
118:         _log.LogWarning("ocorrencia recusada: {Motivo}", motivo);
119:         return BadRequest(new MotivoSaida(motivo));
120:     }
121: }

10: public sealed class ExigeChaveCoordenacaoAttribute : Attribute, IAsyncAuthorizationFilter
11: {
12:     public const string Cabecalho = "X-Chave-Coordenacao";
13: 
14:     public Task OnAuthorizationAsync(AuthorizationFilterContext context)
15:     {
16:         var esperada = context.HttpContext.RequestServices.GetRequiredService<IConfiguration>()["ChaveCoordenacao"] ?? "";
17:         var enviada = context.HttpContext.Request.Headers[Cabecalho].ToString();
18:         var a = Encoding.UTF8.GetBytes(esperada);
19:         var b = Encoding.UTF8.GetBytes(enviada);
20:         if (a.Length == 0 || a.Length != b.Length || !CryptographicOperations.FixedTimeEquals(a, b))
21:             context.Result = new UnauthorizedObjectResult(new MotivoSaida("chave-invalida"));
22:         return Task.CompletedTask;
23:     }
